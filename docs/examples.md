## Exemplos Básicos

### Olá Mundo

```trest
печать("Привет, Мир!")
```

### Calculadora com Switch

```trest
функция калькулятор(a, операция, b) {
    переключатель (операция) {
        случай "+":
            вернуть a + b
        случай "-":
            вернуть a - b
        случай "*":
            вернуть a * b
        случай "/":
            вернуть a / b
        поумолчанию:
            бросить "Неизвестная операция"
    }
}

печать(калькулятор(10, "+", 5))  # 15
печать(калькулятор(10, "*", 3))  # 30
```

### Ternários Simples

```trest
функция абсолютное(число) {
    вернуть число >= 0 ? число : -число
}

печать(абсолютное(5))   # 5
печать(абсолютное(-5))  # 5
```

## Algoritmos

### Fatorial

```trest
функция факториал(n) {
    если (n <= 1) {
        вернуть 1
    }
    вернуть n * факториал(n - 1)
}

печать(факториал(5))  # 120
```

### Fibonacci

```trest
функция фибоначчи(n) {
    если (n <= 1) {
        вернуть n
    }
    вернуть фибоначчи(n - 1) + фибоначчи(n - 2)
}

для (пусть i = 0; i < 10; i = i + 1) {
    печать(фибоначчи(i))
}
```

### Ordenação

```trest
функция пузырек(arr) {
    пусть n = arr.длина
    для (пусть i = 0; i < n - 1; i = i + 1) {
        для (пусть j = 0; j < n - i - 1; j = j + 1) {
            если (arr[j] > arr[j + 1]) {
                пусть temp = arr[j]
                arr[j] = arr[j + 1]
                arr[j + 1] = temp
            }
        }
    }
    вернуть arr
}

пусть числа = [64, 34, 25, 12, 22, 11, 90]
печать(пузырек(числа))
```

## Estruturas de Dados

### Pilha (Stack)

```trest
класс Стек {
    функция конструктор() {
        это.элементы = []
    }
    
    функция добавить(элемент) {
        это.элементы.добавить(элемент)
    }
    
    функция убрать() {
        если (это.элементы.длина == 0) {
            бросить "Стек пуст"
        }
        вернуть это.элементы.удалить(это.элементы.длина - 1)
    }
    
    функция вершина() {
        вернуть это.элементы[это.элементы.длина - 1]
    }
}

пусть стек = новый Стек()
стек.добавить(1)
стек.добавить(2)
стек.добавить(3)
печать(стек.убрать())  # 3
```

### Fila (Queue)

```trest
класс Очередь {
    функция конструктор() {
        это.элементы = []
    }
    
    функция добавить(элемент) {
        это.элементы.добавить(элемент)
    }
    
    функция убрать() {
        если (это.элементы.длина == 0) {
            бросить "Очередь пуста"
        }
        вернуть это.элементы.удалить(0)
    }
}

пусть очередь = новый Очередь()
очередь.добавить("primeiro")
очередь.добавить("segundo")
печать(очередь.убрать())  # "primeiro"
```

## Tarefas Práticas

### Gerenciamento de Usuários

```trest
класс Пользователь {
    функция конструктор(имя, email) {
        это.имя = имя
        это.email = email
        это.id = Математика.случайный()
    }
}

класс МенеджерПользователей {
    функция конструктор() {
        это.пользователи = []
    }
    
    функция добавить(пользователь) {
        это.пользователи.добавить(пользователь)
    }
    
    функция найти(id) {
        для (пусть пользователь из это.пользователи) {
            если (пользователь.id == id) {
                вернуть пользователь
            }
        }
        вернуть нуль
    }
}

пусть менеджер = новый МенеджерПользователей()
пусть иван = новый Пользователь("Иван", "ivan@example.com")
менеджер.добавить(иван)
```

### Calculadora BMI

```trest
функция рассчитатьBMI(вес, рост) {
    вернуть вес / (рост * рост)
}

функция классифицироватьBMI(bmi) {
    если (bmi < 18.5) {
        вернуть "Недостаточный вес"
    } иначе если (bmi < 25) {
        вернуть "Нормальный вес"
    } иначе если (bmi < 30) {
        вернуть "Избыточный вес"
    } иначе {
        вернуть "Ожирение"
    }
}

пусть bmi = рассчитатьBMI(70, 1.75)
печать("BMI: " + bmi)
печать("Категория: " + классифицироватьBMI(bmi))
```
